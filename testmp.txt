import { app, InvocationContext } from "@azure/functions";
import { parseSetpointMessage } from "../messages/parseSetpointMessage";
import { Setpoint } from "../domain/Setpoint";
import { ParseResult } from "../messages/parseSetpointMessage";
import { latestSetPoint } from "../domain/latestSetpoint";
import { isAcceptedDevice } from "../domain/deviceFilter";
import { loadConfig } from "../config";
import { SetpointSender } from "../batteredBatteries/SetpointSender";
import { BatteredBatteriesClient } from "../batteredBatteries/BatteredBatteriesClient";
import { DeliveryOutcome } from "../batteredBatteries/SetpointSender";
import { SetpointProcessorService } from "../Application/SetpointProcessorService";

const config = loadConfig();
const sender = new SetpointSender(
  new BatteredBatteriesClient(config.apiBaseUrl, config.apiKey),
);

export async function setpointConsumer(
  messages: unknown[],
  context: InvocationContext,
): Promise<void> {
  const parsedMessages: Record<string, Setpoint> = {}; 


  for (const message of messages) {
    const result: ParseResult = parseSetpointMessage(message);

    if (!result.isValid) {
      context.log(`Message ${messages} is invalid: ${result.error}`);
      continue;
    }

    if (!isAcceptedDevice(result.value.deviceId)) {
      context.log(
        `The deviceid ${result.value.deviceId} of the ${messages} is not accepted.`,
      );
      continue;
    }

    if (result.value.deviceId in parsedMessages) {
      parsedMessages[result.value.deviceId] = latestSetPoint(
        parsedMessages[result.value.deviceId],
        result.value,
      );
    } else parsedMessages[result.value.deviceId] = result.value;
  }

 const setpointProcessorService=new SetpointProcessorService(sender)


  const setpoints = Object.values(parsedMessages);
  const results = await Promise.all(setpoints.map((s) => sender.send(s)));

  let retryBatch = false;
  for (let i = 0; i < results.length; i++) {
    const { deviceId } = setpoints[i];
    const { outcome, attempts, reason } = results[i];
    const text = `${deviceId}: ${outcome} after ${attempts} attempt(s) (${reason})`;

    if (outcome === DeliveryOutcome.FailedRetryable) {
      context.warn(text);
      retryBatch = true;
    } else if (outcome === DeliveryOutcome.FailedPermanent) {
      context.error(text);
    } else {
      context.log(text);
    }
  }

  // Throwing makes Service Bus redeliver the whole batch later.
  if (retryBatch) {
    throw new Error(
      "One or more setpoints failed temporarily; batch will be retried.",
    );
  }
}

app.serviceBusQueue("setpointConsumer", {
  connection: "CONNECTION-STRING-SBQ-BATBAT-SPT",
  queueName: "sbq-batbat-spt",
  handler: setpointConsumer,
  cardinality: "many",
});
